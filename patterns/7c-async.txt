# 7 clubs, 2 people, the genuinely asynchronous version: B's beats sit half a
# beat after A's, so every throw is a pass and every pass is a single.
#
# The half-beat offset is what forces the fractional heights - a 3 from A would
# arrive between two of B's beats, so the pass has to be a 3.5. This is the same
# thing prechac notation writes as 3.5p.

name: 7c async
beat: 0.30

jugglers:
  A at (0, 1.75)
  B at (0, -1.75) phase 0.5

start:
  A: R=2 L=2
  B: R=2 L=1

loop:
  A: 3.5pB 3.5pB
  B: 3.5pA 3.5pA
